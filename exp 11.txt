
colors = ["Red", "Green", "Blue"]

graph = {
    "A": ["B", "C"],
    "B": ["A", "C", "D"],
    "C": ["A", "B", "D"],
    "D": ["B", "C"]
}

color = {}

def solve(node):
    if node == len(graph):
        return True

    region = list(graph.keys())[node]

    for c in colors:
        safe = True

        for neighbor in graph[region]:
            if color.get(neighbor) == c:
                safe = False
                break

        if safe:
            color[region] = c

            if solve(node + 1):
                return True

            del color[region]

    return False

if solve(0):
    print(color)
else:
    print("No solution")
