
from collections import deque

start = (3, 3, 0)
goal = (0, 0, 1)

moves = [(1, 0), (2, 0), (0, 1),
         (0, 2), (1, 1)]

queue = deque([(start, [])])
visited = {start}

while queue:
    state, path = queue.popleft()
    m, c, boat = state

    if state == goal:
        for s in path + [state]:
            print(s)
        break

    for x, y in moves:
        if boat == 0:
            nm, nc, nb = m-x, c-y, 1
        else:
            nm, nc, nb = m+x, c+y, 0

        if not (0 <= nm <= 3 and 0 <= nc <= 3):
            continue

        rm, rc = 3-nm, 3-nc

        if ((nm == 0 or nm >= nc) and
            (rm == 0 or rm >= rc)):
            new_state = (nm, nc, nb)

            if new_state not in visited:
                visited.add(new_state)
                queue.append((new_state, path + [state]))
else:
    print("No solution found")
