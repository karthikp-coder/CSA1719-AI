
def minimax(depth, node, maximizing, values):
    if depth == 2:
        return values[node]

    if maximizing:
        best = -999

        for i in range(2):
            best = max(best, minimax(depth + 1, node * 2 + i,
                                     False, values))
        return best

    else:
        best = 999

        for i in range(2):
            best = min(best, minimax(depth + 1, node * 2 + i,
                                     True, values))
        return best

values = [3, 5, 2, 9]
print("Best value:", minimax(0, 0, True, values))
